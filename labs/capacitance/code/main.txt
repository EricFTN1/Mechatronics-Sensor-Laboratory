void initComparator(){

void disableCounterMode()
{
    TIMER1_CTL_R &= ~TIMER_CTL_TAEN;                 // turn-off time base timer
    WTIMER1_CTL_R &= ~TIMER_CTL_TAEN;                // turn-off event counter
    NVIC_DIS0_R = 1 << (INT_TIMER1A-16);             // turn-off interrupt 37 (TIMER1A)
}

void enableTimerMode()
{
    TIMER0_CTL_R &= ~(TIMER_CTL_TAEN);               // turn-off timer before reconfiguring
    TIMER0_CFG_R = TIMER_CFG_32_BIT_TIMER;           // configure as 32-bit timer (A+B)
    TIMER0_TAMR_R = TIMER_TAMR_TAMR_PERIOD;          // configure for periodic mode (count down)
    TIMER0_TAILR_R = 40000000;                       // set load value to 40e5 for 40 Hz interrupt rate
    TIMER0_IMR_R = TIMER_IMR_TATOIM;                 // turn-on interrupts
    TIMER0_CTL_R |= TIMER_CTL_TAEN;                  // turn-on timer
    NVIC_EN0_R = 1 << (INT_TIMER0A-16);              // turn-on interrupt 38 (TIMER1B)

    WTIMER1_CTL_R &= ~TIMER_CTL_TAEN;                // turn-off counter before reconfiguring
    WTIMER1_TAILR_R = 0xFFFFFFFF;
    WTIMER1_CFG_R = 4;                               // configure as 32-bit counter (A only)
    WTIMER1_TAMR_R = TIMER_TAMR_TAMR_1_SHOT | TIMER_TAMR_TACDIR;
                                                     // configure for edge time mode, count up
    WTIMER1_TAV_R = 0;                               // zero counter for first period
    WTIMER1_CTL_R |= TIMER_CTL_TAEN;                 // turn-on counter
}

// Initialize Hardware
void initHw()
{
    // Initialize system clock to 40 MHz
    initSystemClockTo40Mhz();

    // Enable clocks
    SYSCTL_RCGCTIMER_R |= (SYSCTL_RCGCTIMER_R4 | SYSCTL_RCGCTIMER_R3 | SYSCTL_RCGCTIMER_R2 | SYSCTL_RCGCTIMER_R1 | SYSCTL_RCGCTIMER_R0);
    SYSCTL_RCGCWTIMER_R |= SYSCTL_RCGCWTIMER_R1;
    SYSCTL_RCGCGPIO_R |= SYSCTL_RCGCGPIO_R2 | SYSCTL_RCGCGPIO_R5 | SYSCTL_RCGCGPIO_R4;

    _delay_cycles(3);

    SYSCTL_RCGCGPIO_R |= SYSCTL_RCGCGPIO_R4;
    _delay_cycles(3);

    // Configure AIN3 as an analog input
    GPIO_PORTC_AFSEL_R |= 128;                 // select alternative functions for AIN3 (PE0)
    GPIO_PORTC_DEN_R &= ~128;                  // turn off digital operation on pin PE0
    GPIO_PORTC_AMSEL_R |= 128;                 // turn on analog operation on pin PE0
    GPIO_PORTC_DIR_R &= ~128;

    GPIO_PORTE_DEN_R |= 2;
    GPIO_PORTE_PDR_R |= 2;
    GPIO_PORTE_DIR_R |= 2;

    // Initialize system clock to 40 MHz
    _delay_cycles(3);
    initUart0();
    initComparator();
    setUart0BaudRate(115200,40000000); // Set baud rate to project spec
}
